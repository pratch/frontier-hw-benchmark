\begin{frame}{The Scorer: Relaxed Accuracy (No LLM Judge)}
    \begin{columns}[T]
        \begin{column}{0.55\textwidth}
            \begin{block}{\textbf{Why Avoid LLM-as-a-Judge?}}
                \begin{itemize}
                    \item Fulfills assignment rule: \textit{``checked by program, no judge to validate and no API bill''}.
                    \item 100\% deterministic, reproducible, and verifiable in pure Python.
                \end{itemize}
            \end{block}

            \vspace{0.2cm}
            \begin{block}{\textbf{Metric Definition (Methani et al., 2020)}}
                \begin{itemize}
                    \item \textbf{Numeric Answers}: Allows $\mathbf{\le 5\%}$ relative error tolerance to accommodate visual estimation:
                    $$\text{Correct} \iff \frac{|y_{\text{pred}} - y_{\text{true}}|}{\max(|y_{\text{true}}|, 10^{-9})} \le 0.05$$
                    \item \textbf{Text / Category}: Normalized case-insensitive exact string match (punctuation-stripped).
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.42\textwidth}
            \begin{tcolorbox}[title=\textbf{Scorer Behavior Examples}]
                \small
                \begin{tabular}{@{}llcc@{}}
                    \toprule
                    \textbf{Target} & \textbf{Prediction} & \textbf{Rel. $\Delta$} & \textbf{Result} \\
                    \midrule
                    \texttt{100} & \texttt{97} & 3.0\% & \textcolor{success}{\ding{51} Correct} \\
                    \texttt{100} & \texttt{90} & 10.0\% & \textcolor{red}{\ding{55} False} \\
                    \texttt{45\%} & \texttt{44.8\%} & 0.4\% & \textcolor{success}{\ding{51} Correct} \\
                    \texttt{Bar chart} & \texttt{bar chart.} & 0.0\% & \textcolor{success}{\ding{51} Correct} \\
                    \bottomrule
                \end{tabular}
                \vspace{0.2cm}
                
                \footnotesize
                \textit{Unit tested in \texttt{test\_scorer.py} with 100\% passing test vectors.}
            \end{tcolorbox}
        \end{column}
    \end{columns}
\end{frame}
